```ts
import { RabbitMQBroker, event } from "@bitspacerlabs/rabbit-relay";

const broker = new RabbitMQBroker("orders_service");

const api = await broker
  .queue("orders.q")
  .exchange<{ orderCreated: EventEnvelope<OrderCreated> }>("orders.ex", {
    exchangeType: "topic",
    routingKey: "orders.*",
    publisherConfirms: true,
  })
  .with({ orderCreated });

api.handle("orderCreated", async (_id, ev) => {
  await fulfill(ev.data);
});

await api.consume({ prefetch: 20 });
```
